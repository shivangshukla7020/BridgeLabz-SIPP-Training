Moderate – Patient Record Hashing

class Patient {
    int id;
    String name;
    String illness;
}

class PatientManager {
    HashMap<Integer, Patient> records;

    void addPatient(Patient p) -> records.put(p.id, p);
    void updatePatient(Patient p) -> records.put(p.id, p);
    void removePatient(int id) -> records.remove(id);
    Patient getPatient(int id) -> return records.get(id);
}

Moderate – Maze Solver Bot

class MazeSolver {
    int[][] maze;  // 0 = path, 1 = wall
    boolean[][] visited;

    boolean solve(int x, int y) {
        if (x,y is exit) return true
        if (x,y is invalid or visited) return false

        mark visited[x][y] = true
        if solve(x+1,y) or solve(x-1,y) or solve(x,y+1) or solve(x,y-1)
            return true

        backtrack
        return false
    }
}


Moderate – Library Book Filter

class Book {
    String title;
    int year;
    double rating;
}

class Library {
    List<Book> books;

    void filterBooks() {
        books.stream()
            .filter(b -> b.year > 2015)
            .filter(b -> b.rating > 4.5)
            .forEach(print each book);
    }
}

Moderate – Payment Processor

class Transaction<T> {
    T data;
    double amount;

    void process() {
        print "Processing transaction of " + amount + " with data: " + data
    }
}

class PaymentProcessor {
    <T> void handleTransaction(Transaction<T> tx) {
        tx.process();
    }
}
