Easy – Traffic Signal Simulation

enum TrafficLight { RED, GREEN, YELLOW }

class TrafficSignal {
    TrafficLight current;

    void switchLight() {
        if (current == RED) set GREEN;
        else if (current == GREEN) set YELLOW;
        else if (current == YELLOW) set RED;
    }

    void runSimulation() {
        loop forever {
            print current light
            wait few seconds
            switchLight()
        }
    }
}

Moderate – Job Scheduler

class Job {
    int deadline;
    int duration;
}

class JobScheduler {
    PriorityQueue<Job> pq; // priority by earliest deadline

    void addJob(Job j) -> pq.add(j);
    void executeJobs() {
        while !pq.empty() {
            Job j = pq.remove();
            process job j
        }
    }
}

Easy – Online Store Cart System

class Item {
    String name;
    double price;
}

class Cart {
    List<Item> items;

    <T extends Item> double totalPrice(List<T> list) {
        double total = 0;
        for (T item : list)
            total += item.price;
        return total;
    }
}
