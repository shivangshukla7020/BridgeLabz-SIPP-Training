Moderate – Banking Portal

class Account {
    String accountNumber;
    double balance;

    void deposit(double amount) {
        balance += amount;
    }

    void withdraw(double amount) {
        // generic withdrawal (to be overridden)
    }
}

class SavingAccount extends Account {
    @Override
    void withdraw(double amount) {
        if (balance - amount >= 500) {
            balance -= amount;
            print "Withdrawn " + amount;
        } else {
            print "Insufficient balance. Minimum 500 required";
        }
    }
}

class CurrentAccount extends Account {
    @Override
    void withdraw(double amount) {
        if (balance >= amount) {
            balance -= amount;
            print "Withdrawn " + amount;
        } else {
            print "Overdraft not allowed";
        }
    }
}

Difficult – Vehicle Rental System

interface Rentable {
    void rent();
}

abstract class Vehicle implements Rentable {
    String model;
    int year;
}

class Car extends Vehicle {
    void rent() -> print "Car rented: " + model;
}

class Bike extends Vehicle {
    void rent() -> print "Bike rented: " + model;
}

class VehicleFactory {
    static Vehicle createVehicle(String type) {
        if (type == "car") return new Car();
        if (type == "bike") return new Bike();
        return null;
    }
}

Moderate – College Course Registration

class Student {
    String name;
    int rollNo;
}

class Course {
    String code;
    String title;
}

class College {
    Map<Course, List<Student>> registrations;

    void register(Course c, Student s) {
        registrations.putIfAbsent(c, new ArrayList<>());
        registrations.get(c).add(s);
    }

    void display() {
        for each course in registrations.keySet():
            print course.title + " -> " + registrations.get(course);
    }
}

Difficult – Digital Wallet Transactions Report

class Transaction {
    String merchant;
    double amount;
}

class ReportGenerator {
    void generate(List<Transaction> txs) {
        Map<String, List<Transaction>> grouped =
            txs.stream().collect(Collectors.groupingBy(t -> t.merchant));

        grouped.forEach((merchant, list) -> {
            double total = list.stream().mapToDouble(t -> t.amount).sum();
            print merchant + " total = " + total;
        });
    }
}
