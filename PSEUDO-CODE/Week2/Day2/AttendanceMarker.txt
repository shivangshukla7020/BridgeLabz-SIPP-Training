CLASS AttendanceSystem
    INTEGER[][] attendance   // rows = students, cols = days

    METHOD markAttendance(student, day, status)
        attendance[student][day] = status   // 1 = Present, 0 = Absent

    METHOD dailySummary(day)
        count = 0
        FOR each student in attendance
            IF attendance[student][day] == 1
                count++
        PRINT "Students present on Day", day, "=", count

    METHOD monthlySummary(student)
        total = 0
        FOR each day in attendance[student]
            total += attendance[student][day]
        PRINT "Student", student, "present days =", total
